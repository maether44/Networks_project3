FAULT INJECTION - turns Team4_working.pkt into Team4_broken.pkt

1. Open Team4_working.pkt and do File > Save As Team4_broken.pkt FIRST.
2. Paste each block into that device's CLI.
3. Change POS2's gateway by hand (step F4).
4. Wait about 1 minute, check each symptom from docs/Team4_answer_key.md, then File > Save.

--- F1: SW2 (wrong access VLAN for POS3) ---
enable
configure terminal
interface FastEthernet0/11
 switchport access vlan 30
end
write memory

--- F2: SW1 (VLAN 10 removed from trunk to SW2) ---
enable
configure terminal
interface GigabitEthernet0/2
 switchport trunk allowed vlan 20,30
end
write memory

--- F3: R1 (Management LAN removed from OSPF) ---
enable
configure terminal
router ospf 1
 no network 192.168.40.0 0.0.0.255 area 0
end
write memory

--- F4: POS2 (wrong default gateway) ---
POS2 > Desktop > IP Configuration > Default Gateway = 192.168.20.254

--- F5: R2 (guest ACL too permissive) ---
enable
configure terminal
no ip access-list extended GUEST-IN
ip access-list extended GUEST-IN
 permit udp any any eq bootps
 permit icmp 192.168.30.0 0.0.0.255 host 192.168.30.1
 permit ip 192.168.30.0 0.0.0.255 192.168.50.0 0.0.0.255
 deny ip any any
end
write memory

--- F6: R3 (OSPF hello mismatch on link to R2) ---
enable
configure terminal
interface GigabitEthernet0/1
 ip ospf hello-interval 5
! PT 8.2.2 keeps dead-interval 40; only the hello timers differ
end
write memory
